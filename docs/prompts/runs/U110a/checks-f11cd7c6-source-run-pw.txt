# Check-runner wrapper (this check only, not committed): runs one Playwright command from app/ detached,
# keeps the raw log and the exit code. Mirrors U110's run-sweep.ps1 / run-e2e.ps1.
# Usage: powershell -File run-pw.ps1 -Name <log name> -Config <config> -Workers 2 -Specs "<space separated spec filters and flags>" [-Probes 1] [-Out <U110_OUT>]
param([string]$Name, [string]$Config, [int]$Workers = 2, [string]$Specs = '', [string]$Probes = '', [string]$Out = '', [string]$Extra = '')
$here = $PSScriptRoot
$app = Join-Path $here '..\..\app'
Set-Location $app
if ($Probes -eq '1') { $env:U110_PROBES = '1' }
if ($Out -ne '') { $env:U110_OUT = $Out }
if ($Extra -ne '') { foreach ($kv in $Extra.Split(';')) { $p = $kv.Split('=', 2); Set-Item -Path ("Env:" + $p[0]) -Value $p[1] } }
$logPath = Join-Path $here "$Name.log"
$list = @()
if ($Specs -ne '') { $list = $Specs.Split(' ') }
$start = Get-Date
& npx playwright test --config $Config @list --workers=$Workers *> $logPath
$code = $LASTEXITCODE
$end = Get-Date
Set-Content -Path (Join-Path $here "$Name.times") -Value "start $($start.ToString('o')) end $($end.ToString('o'))" -Encoding ascii
Set-Content -Path (Join-Path $here "$Name.exit") -Value $code -Encoding ascii
