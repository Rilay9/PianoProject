// Prints the admission trace as pricing passes (groups of four: the chosen shape, then the candidates for
// shown = 3, 2, 1) per load, with derived columns. Usage: node admission-table.cjs <json> [all|last]
const fs = require('node:fs');
const [file, mode = 'all'] = process.argv.slice(2);
const r1 = (x) => Math.round(x * 10) / 10;
for (const l of JSON.parse(fs.readFileSync(file, 'utf8'))) {
  const adm = l.log.filter((e) => e && e.kind === 'admission');
  const okShape = adm.length % 4 === 0 && adm.every((e, i) => i % 4 === 0 || e.shown === 4 - (i % 4));
  console.log(`## ${l.load}: ${adm.length} entries = ${adm.length / 4} passes of 4; group shape verified: ${okShape}; end state slots ${l.slots} systems ${l.systems} shown ${l.shown} ahead ${l.ahead} ladder ${JSON.stringify(l.shapeChanges)} overlapPx ${l.overlapPx}`);
  const groups = [];
  for (let i = 0; i < adm.length; i += 4) groups.push(adm.slice(i, i + 4));
  groups.forEach((g, gi) => {
    if (mode === 'last' && gi < groups.length - 1) return;
    const c = g[0];
    const gap = (c.admissionPx - c.systems * c.rowHeight - c.aheadHeight) / c.systems;
    console.log(
      `  pass ${String(gi).padStart(2)} chosen shown ${c.shown} systems ${c.systems} maxSlots ${c.maxSlots} ahead ${c.ahead} | rowHeight ${r1(c.rowHeight)} aheadHeight ${r1(c.aheadHeight)} admissionPx ${r1(c.admissionPx)} slotsHeight ${r1(c.slotsHeight)} (admission - room ${r1(c.admissionPx - c.slotsHeight)}) | drawnHere ${c.drawnHere} drawnNow ${c.drawnNow} drawnRowPx ${r1(c.drawnRowPx)} currentSlots ${c.currentSlots} zoom ${c.zoom} | gap/system ${r1(gap)}`,
    );
  });
}
