// Classifies every pricing pass (group of four: chosen shape, then candidates for shown 3, 2, 1) of the admission
// trace, per load. Usage: node admission-classify.cjs <json>
// refused-by-room: ahead false although systems + 1 <= maxSlots and admissionPx > slotsHeight (the third clause of
//   aheadFor, windowAt(...) < pieceBars - 1, is not logged; Twinkle's window is at its start, bars 1-3 of 12).
// held: a look-ahead row is currently drawn (currentSlots > systems) and the chosen shape has none (ahead false).
// followed-by: the next pass's currentSlots (did the shape change after this pass, or stay).
// windowOnlyDrawnRowPx: drawnRowPx at the latest earlier pass of this load at the same zoom with currentSlots == systems.
// altPx: the same sum with the window rows priced at windowOnlyDrawnRowPx instead of the reserve rowHeight.
const fs = require('node:fs');
const r1 = (x) => Math.round(x * 10) / 10;
const file = process.argv[2];
for (const l of JSON.parse(fs.readFileSync(file, 'utf8'))) {
  const adm = l.log.filter((e) => e && e.kind === 'admission');
  const groups = [];
  for (let i = 0; i < adm.length; i += 4) groups.push(adm.slice(i, i + 4));
  console.log(`## ${l.load}: ${groups.length} passes; end slots ${l.slots}/${l.systems}/${l.shown}, ladder record ${JSON.stringify(l.shapeChanges)}`);
  const winOnly = new Map();
  let appliedChanges = 0;
  groups.forEach((g, i) => {
    const c = g[0];
    if (c.currentSlots === c.systems && c.drawnHere && c.drawnRowPx > 0) winOnly.set(c.zoom, Math.min(winOnly.get(c.zoom) ?? Infinity, c.drawnRowPx));
    const next = groups[i + 1] ? groups[i + 1][0].currentSlots : null;
    const answerSlots = c.ahead ? c.systems + 1 : c.systems;
    const refusedByRoom = !c.ahead && c.systems + 1 <= c.maxSlots && c.admissionPx > c.slotsHeight;
    const held = c.currentSlots > c.systems && !c.ahead;
    const gap = (c.admissionPx - c.systems * c.rowHeight - c.aheadHeight) / c.systems;
    const w = winOnly.get(c.zoom);
    const alt = w ? c.systems * Math.min(c.rowHeight, w) + c.aheadHeight + gap * c.systems : null;
    const changed = answerSlots !== c.currentSlots;
    const stayed = next !== null && next === c.currentSlots;
    if (changed && next !== null && next !== c.currentSlots) appliedChanges += 1;
    const tag = [c.ahead ? 'granted' : refusedByRoom ? 'refused-by-room' : 'refused-other', held ? 'HELD-ROW-DRAWN' : '', changed ? (stayed ? 'answer-differs-shape-stayed' : 'answer-differs-shape-changed') : ''].filter(Boolean).join(' ');
    if (held || (!c.ahead && c.currentSlots === c.systems && refusedByRoom && c.admissionPx - c.slotsHeight < 20)) {
      console.log(
        `  pass ${String(i).padStart(2)} zoom ${c.zoom} room ${r1(c.slotsHeight)} admissionPx ${r1(c.admissionPx)} (over by ${r1(c.admissionPx - c.slotsHeight)}) rowHeight ${r1(c.rowHeight)} aheadHeight ${r1(c.aheadHeight)} drawnRowPx ${r1(c.drawnRowPx)} windowOnlyDrawnRowPx ${w ? r1(w) : '-'} altPx ${alt ? r1(alt) : '-'} (${alt ? (alt <= c.slotsHeight ? 'fits' : 'does not fit') : '-'}) currentSlots ${c.currentSlots} next ${next} | ${tag}`,
      );
    }
  });
  console.log(`  applied shape changes seen between passes: ${appliedChanges}`);
}
